   264→   264→export function LiveTvView() {
   265→   265→  useReveal();
   266→   266→  const [category, setCategory] = useState<string>("all");
   267→   267→  const [active, setActive] = useState<Channel>(CHANNELS[0]);
   268→   268→  const [serverIdx, setServerIdx] = useState(0);
   269→   269→  const [playing, setPlaying] = useState(false);
   270→   270→  const [muted, setMuted] = useState(true);
   271→   271→  const [loading, setLoading] = useState(false);
   272→   272→  const [error, setError] = useState<string | null>(null);
   273→   273→  const [showCustom, setShowCustom] = useState(false);
   274→   274→  const [customUrl, setCustomUrl] = useState("");
   275→   275→  const [customName, setCustomName] = useState("");
   276→   276→  const videoRef = useRef<HTMLVideoElement>(null);
   277→   277→  const hlsRef = useRef<Hls | null>(null);
   278→   278→
   279→   279→  const filtered = useMemo(
   280→   280→    () => (category === "all" ? CHANNELS : CHANNELS.filter((c) => c.category === category)),
   281→   281→    [category]
   282→   282→  );
   283→   283→
   284→   284→  const src = active.servers[Math.min(serverIdx, active.servers.length - 1)].url;
   285→   285→
   286→   286→  const destroyHls = useCallback(() => {
   287→   287→    if (hlsRef.current) {
   288→   288→      hlsRef.current.destroy();
   289→   289→      hlsRef.current = null;
   290→   290→    }
   291→   291→  }, []);
   292→   292→
   293→   293→  const loadStream = useCallback(
   294→   294→    async (url: string) => {
   295→   295→      const video = videoRef.current;
   296→   296→      if (!video) return;
   297→   297→      destroyHls();
   298→   298→      setError(null);
   299→   299→      setLoading(true);
   300→   300→      try {
   301→   301→        if (video.canPlayType("application/vnd.apple.mpegurl") && url.includes(".m3u8") && !url.includes("test-streams")) {
   302→   302→          // Native HLS (Safari / iOS)
   303→   303→          video.src = url;
   304→   304→        } else {
   305→   305→          const HlsCtor = (await import("hls.js")).default;
   306→   306→          if (HlsCtor.isSupported()) {
   307→   307→            const hls = new HlsCtor({ enableWorker: true, lowLatencyMode: false, backBufferLength: 90 });
   308→   308→            hlsRef.current = hls;
   309→   309→            hls.loadSource(url);
   310→   310→            hls.attachMedia(video);
   311→   311→            hls.on(HlsCtor.Events.ERROR, (_e, data) => {
   312→   312→              if (data.fatal) {
   313→   313→                setLoading(false);
   314→   314→                setError("This stream is currently unavailable. Try another server or channel.");
   315→   315→                destroyHls();
   316→   316→              }
   317→   317→            });
   318→   318→          } else {
   319→   319→            video.src = url;
   320→   320→          }
   321→   321→        }
   322→   322→        video.muted = muted;
   323→   323→        await video.play().catch(() => {
   324→   324→          /* autoplay may need a manual click */
   325→   325→        });
   326→   326→        setPlaying(!video.paused);
   327→   327→      } finally {
   328→   328→        setLoading(false);
   329→   329→      }
   330→   330→    },
   331→   331→    [destroyHls, muted]
   332→   332→  );
   333→   333→
   334→   334→  // Load stream when channel or server changes (only if user has started playback)
   335→   335→  useEffect(() => {
   336→   336→    if (playing || error) void loadStream(src);
   337→   337→  }, [src]);
   338→   338→
   339→   339→  useEffect(() => () => destroyHls(), [destroyHls]);
   340→   340→
   341→   341→  const startPlayback = (ch: Channel, idx = 0) => {
   342→   342→    setActive(ch);
   343→   343→    setServerIdx(idx);
   344→   344→    setPlaying(true);
   345→   345→    setError(null);
   346→   346→    // stream loads via the src-change effect
   347→   347→  };
   348→   348→
   349→   349→  const togglePlay = async () => {
   350→   350→    const video = videoRef.current;
   351→   351→    if (!video) return;
   352→   352→    if (!playing && !hlsRef.current && !video.src) {
   353→   353→      await loadStream(src);
   354→   354→      return;
   355→   355→    }
   356→   356→    if (video.paused) {
   357→   357→      await video.play().catch(() => {});
   358→   358→      setPlaying(true);
   359→   359→    } else {
   360→   360→      video.pause();
   361→   361→      setPlaying(false);
   362→   362→    }
   363→   363→  };
   364→   364→
   365→   365→  const toggleMute = () => {
   366→   366→    const video = videoRef.current;
   367→   367→    if (!video) return;
   368→   368→    video.muted = !video.muted;
   369→   369→    setMuted(video.muted);
   370→   370→  };
   371→   371→
   372→   372→  const goFullscreen = () => {
   373→   373→    const el = videoRef.current?.parentElement;
   374→   374→    if (el?.requestFullscreen) void el.requestFullscreen();
   375→   375→  };
   376→   376→
   377→   377→  const playCustom = () => {
   378→   378→    const url = customUrl.trim();
   379→   379→    if (!/^https?:\/\/.+/i.test(url)) return;
   380→   380→    const ch: Channel = {
   381→   381→      id: "custom",
   382→   382→      name: customName.trim() || "Custom Stream",
   383→   383→      monogram: "CS",
   384→   384→      color: "#006A4E",
   385→   385→      category: "sports",
   386→   386→      tagline: "User-provided source",
   387→   387→      nowPlaying: "External stream",
   388→   388→      next: "—",
   389→   389→      viewers: 1,
   390→   390→      demo: false,
   391→   391→      servers: [{ label: "Custom", url, note: "User source" }],
   392→   392→    };
   393→   393→    setShowCustom(false);
   394→   394→    startPlayback(ch, 0);
   395→   395→  };
   396→   396→
   397→   397→  return (
   398→   398→    <div className="mx-auto w-full max-w-7xl px-4 pb-14 pt-6 sm:px-6">
   399→   399→      <PageHeader
   400→   400→        title="Live TV"
   401→   401→        kicker="multi-server streaming"
   402→   402→        subtitle="Sports and football channels streamed from multiple servers — switch sources instantly, never miss kick-off"
   403→   403→        icon={<Tv className="h-5 w-5" aria-hidden />}
   404→   404→      >
   405→   405→        <span className="flex items-center gap-1.5 rounded-full border border-fhb-red/45 bg-fhb-red/10 px-3 py-1.5 text-[10px] font-bold text-fhb-red">
   406→   406→          <span className="live-dot h-1.5 w-1.5 rounded-[2px] bg-fhb-red" aria-hidden />
   407→   407→          {fmtNum(CHANNELS.length)} channels online
   408→   408→        </span>
   409→   409→      </PageHeader>
   410→   410→
   411→   411→      {/* Billboard ad */}
   412→   412→      <div className="reveal mt-6">
   413→   413→        <AdSlot placement="HEADER_BANNER" />
   414→   414→      </div>
   415→   415→
   416→   416→      <div className="mt-6 grid gap-6 lg:grid-cols-3">
   417→   417→        {/* ═══ Player ═══ */}
   418→   418→        <section className="min-w-0 lg:col-span-2" aria-label="Live stream player">
   419→   419→          <div className="edge-lit overflow-hidden rounded-2xl border border-border bg-card">
   420→   420→            {/* Video area */}
   421→   421→            <div className="relative aspect-video w-full bg-black">
   422→   422→              <video
   423→   423→                ref={videoRef}
   424→   424→                className="h-full w-full max-w-full object-contain"
   425→   425→                playsInline
   426→   426→                controls={false}
   427→   427→                onError={() => {
   428→   428→                  if (playing) setError("This stream is currently unavailable. Try another server or channel.");
   429→   429→                }}
   430→   430→                aria-label={`Live stream: ${active.name}`}
   431→   431→              />
   432→   432→              {/* Idle state */}
   433→   433→              {!playing && !error && (
   434→   434→                <button
   435→   435→                  onClick={() => void loadStream(src)}
   436→   436→                  className="group absolute inset-0 flex flex-col items-center justify-center gap-4 bg-gradient-to-b from-black/60 via-black/40 to-black/70"
   437→   437→                  aria-label={`Play ${active.name}`}
   438→   438→                >
   439→   439→                  <span className="flex h-20 w-20 items-center justify-center rounded-full border-2 border-white/25 bg-black/50 backdrop-blur-md transition-transform duration-300 group-hover:scale-110">
   440→   440→                    <Play className="ml-1 h-9 w-9 fill-white text-white" />
   441→   441→                  </span>
   442→   442→                  <span className="text-center">
   443→   443→                    <span className="block font-display text-lg font-bold tracking-wide text-white">
   444→   444→                      {active.name}
   445→   445→                    </span>
   446→   446→                    <span className="mt-1 block text-xs text-white/70">
   447→   447→                      {active.nowPlaying} · tap to start the stream
   448→   448→                    </span>
   449→   449→                  </span>
   450→   450→                </button>
   451→   451→              )}
   452→   452→              {/* Loading */}
