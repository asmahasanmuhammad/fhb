   349→  homeTeam: { id: string; name: string; initials: string; color: string };

   353→
   354→function MatchManager() {
   355→  const { toast } = useToast();
   356→  const queryClient = useQueryClient();
   357→
   358→  const { data, isLoading } = useQuery<{ matches: AdminMatch[] }>({
   359→    queryKey: ["admin", "matches"],
   360→    queryFn: async () => {
   361→      const res = await fetch("/api/admin/matches");
   362→      return res.json().then((j) => j.data);
   363→    },
   364→  });
   365→
   366→  const update = useMutation({
   367→    mutationFn: async (payload: Record<string, unknown>) => {
   368→      const res = await fetch("/api/admin/matches", {
   369→        method: "PUT",
   370→        headers: { "Content-Type": "application/json" },
   371→        body: JSON.stringify(payload),
   372→      });
   373→      const json = await res.json();
   374→      if (!json.success) throw new Error(json.error);
   375→      return json;
   376→    },
   377→    onSuccess: (_d, payload) => {
   378→      const msg =
   379→        payload.status === "LIVE"
   380→          ? "🔴 ম্যাচ এখন লাইভ! লাইভ-সিমুলেশন সার্ভিস এটি ১৫ সেকেন্ডের মধ্যে নিয়ে নেবে।"
   381→          : "ম্যাচ আপডেট হয়েছে ✅";
   382→      toast({ title: msg });
   383→      queryClient.invalidateQueries({ queryKey: ["admin", "matches"] });
   384→      queryClient.invalidateQueries({ queryKey: ["matches"] });
   385→    },
   386→    onError: (e: Error) => toast({ title: "আপডেট ব্যর্থ", description: e.message, variant: "destructive" }),
   387→  });
   388→
   389→  const matches = data?.matches ?? [];
   390→  const sorted = [...matches].sort((a, b) => {
   391→    const rank = (s: string) => (s === "LIVE" ? 0 : s === "SCHEDULED" ? 1 : 2);
   392→    return rank(a.status) - rank(b.status) || +new Date(b.kickoff) - +new Date(a.kickoff);
   393→  });
   394→
   395→  return (
