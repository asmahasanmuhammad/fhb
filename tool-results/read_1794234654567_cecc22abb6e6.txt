     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import { useQuery } from "@tanstack/react-query";
     5→import { useAppStore } from "@/store/app-store";
     6→import { useLiveScores } from "@/components/providers/live-score-provider";
     7→import { useReveal } from "@/hooks/use-reveal";
     8→import type { Match, MatchDetail } from "@/lib/types";
     9→import { MatchRow } from "@/components/football/match-card";
    10→import { TeamCrest } from "@/components/football/team-badge";
    11→import { LiveBadge } from "@/components/football/live-badge";
    12→import { AdSlot } from "@/components/football/ad-slot";
    13→import { PageHeader, SectionHeader } from "@/components/football/section-header";
    14→import { Skeleton } from "@/components/ui/skeleton";
    15→import { cn } from "@/lib/utils";
    16→import { formatKickoff, fmtNum } from "@/lib/format";
    17→import {
    18→  Activity, ArrowLeft, ArrowLeftRight, CalendarDays, CalendarX, ChevronRight, Clock,
    19→  FileText, Flag, Goal, ListOrdered, MapPin, Table2, Timer, Trophy, Tv, Zap,
    20→} from "lucide-react";
    21→
    22→/* ════════════════════════════════════════════════════════════════
    23→   ScoresView — FotMob-style match centre
    24→   Date rail + status tabs, competition-grouped dense rows, side rail.
    25→   ════════════════════════════════════════════════════════════════ */
    26→
    27→type Tab = "all" | "live" | "upcoming" | "finished";
    28→
    29→const DAY_MS = 86_400_000;
    30→const DOW_SHORT = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
    31→/** How far back/forward the date rail scans for fixture days (matches the API window). */
    32→const RAIL_SPAN_DAYS = 30;
    33→/** Right-rail quick links, resolved against competitions present in the data. */
    34→const TOP_COMPETITIONS = ["EPL", "UCL", "LaLiga", "Serie A", "Bundesliga", "BFL"];
    35→const STATUS_TABS: Tab[] = ["all", "live", "upcoming", "finished"];
    36→const TAB_LABELS: Record<Tab, string> = {
    37→  all: "All",
    38→  live: "Live",
    39→  upcoming: "Upcoming",
    40→  finished: "Finished",
    41→};
    42→
    43→export function ScoresView() {
    44→  const [userTab, setUserTab] = useState<Tab | null>(null);
    45→  const [dayOffset, setDayOffset] = useState(0); // 0 = today
    46→  const live = useLiveScores();
    47→  const liveConnected = live.connected;
    48→  const navigate = useAppStore((s) => s.navigate);
    49→  useReveal();
    50→
    51→  // Track the site header height so the sub-nav sticks exactly below it —
    52→  // the header grows with the live ticker row and the desktop league rail.
    53→  const [headerH, setHeaderH] = useState<number | null>(null);
    54→  useEffect(() => {
    55→    const header = document.querySelector("header");
    56→    if (!header) return;
    57→    const update = () => setHeaderH(Math.round(header.getBoundingClientRect().height));
    58→    update();
    59→    const ro = new ResizeObserver(update);
    60→    ro.observe(header);
    61→    return () => ro.disconnect();
    62→  }, []);
    63→
    64→  const { data, isLoading } = useQuery<{ matches: Match[] }>({
    65→    queryKey: ["matches", "scores"],
    66→    queryFn: async () => {
    67→      const res = await fetch("/api/matches?status=ALL&days=60&limit=150");
    68→      return res.json().then((j) => j.data);
    69→    },
    70→    refetchInterval: 60_000, // fallback refresh for socket-less situations
    71→  });
    72→
    73→  /** Fetched matches with any socket live-state merged in. */
    74→  const matches = useMemo(() => {
    75→    const all = data?.matches ?? [];
    76→    const liveIds = new Set(live.matches.map((m) => m.id));
    77→    return all.map((m) => (liveIds.has(m.id) ? { ...m, status: "LIVE" as const } : m));
    78→  }, [data, live.matches]);
    79→
    80→  const liveMatches = useMemo(() => matches.filter((m) => m.status === "LIVE"), [matches]);
    81→  const liveCount = liveMatches.length;
    82→
    83→  // Smart default tab: land on "live" only when matches are actually live
    84→  const tab: Tab = userTab ?? (liveCount > 0 ? "live" : "all");
    85→
    86→  /** Date rail chips — Yesterday/Today/Tomorrow anchors plus every day with
    87→   *  fixtures inside the fetched window, so no match is unreachable. */
    88→  const dayChips = useMemo(() => {
    89→    const base = new Date();
    90→    base.setHours(0, 0, 0, 0);
    91→    const counts = new Map<number, number>();
    92→    for (const m of matches) {
    93→      const d = new Date(m.kickoff);
    94→      d.setHours(0, 0, 0, 0);
    95→      const key = Math.round((d.getTime() - base.getTime()) / DAY_MS);
    96→      counts.set(key, (counts.get(key) ?? 0) + 1);
    97→    }
    98→    const chips: Array<{ offset: number; count: number; label: string; fullLabel: string }> = [];
    99→    const push = (offset: number) => {
   100→      const d = new Date(base.getTime() + offset * DAY_MS);
   101→      chips.push({
   102→        offset,
   103→        count: counts.get(offset) ?? 0,
   104→        label:
   105→          offset === -1
   106→            ? "Yesterday"
   107→            : offset === 0
   108→              ? "Today"
   109→              : offset === 1
   110→                ? "Tomorrow"
   111→                : `${DOW_SHORT[d.getDay()]} ${d.getDate()}`,
   112→        fullLabel: formatKickoff(d.toISOString()).date,
   113→      });
   114→    };
   115→    // Past fixture days, then the three anchors, then upcoming fixture days
   116→    for (let o = -RAIL_SPAN_DAYS; o <= -2; o++) if ((counts.get(o) ?? 0) > 0) push(o);
   117→    for (const o of [-1, 0, 1]) push(o);
   118→    for (let o = 2; o <= RAIL_SPAN_DAYS; o++) if ((counts.get(o) ?? 0) > 0) push(o);
   119→    return chips;
   120→  }, [matches]);
   121→
   122→  const selectedDay = dayChips.find((c) => c.offset === dayOffset) ?? dayChips.find((c) => c.offset === 0) ?? dayChips[0];
   123→
   124→  /** Matches on the selected day (live state already merged). */
   125→  const dayMatches = useMemo(() => {
   126→    const base = new Date();
   127→    base.setHours(0, 0, 0, 0);
   128→    const target = base.getTime() + dayOffset * DAY_MS;
   129→    return matches.filter((m) => {
   130→      const d = new Date(m.kickoff);
   131→      d.setHours(0, 0, 0, 0);
   132→      return d.getTime() === target;
   133→    });
   134→  }, [matches, dayOffset]);
   135→
   136→  /** Rows for the active tab. Live is global — live play is always "now". */
   137→  const pool = useMemo(() => {
   138→    if (tab === "live") return liveMatches;
   139→    if (tab === "upcoming") return dayMatches.filter((m) => m.status === "SCHEDULED");
   140→    if (tab === "finished") return dayMatches.filter((m) => m.status === "FINISHED");
   141→    return dayMatches;
   142→  }, [tab, liveMatches, dayMatches]);
   143→
   144→  /** Group by competition; live competitions float to the top. */
   145→  const grouped = useMemo(() => {
   146→    const byComp = new Map<string, { comp: Match["competition"]; list: Match[] }>();
   147→    for (const m of pool) {
   148→      const g = byComp.get(m.competition.id) ?? { comp: m.competition, list: [] };
   149→      g.list.push(m);
   150→      byComp.set(m.competition.id, g);
   151→    }
   152→    const rank = (m: Match) => (m.status === "LIVE" ? 0 : m.status === "SCHEDULED" ? 1 : 2);
   153→    const groups = [...byComp.values()];
   154→    for (const g of groups) {
   155→      g.list.sort((a, b) => {
   156→        const r = rank(a) - rank(b);
   157→        if (r !== 0) return r;
   158→        const ka = +new Date(a.kickoff);
   159→        const kb = +new Date(b.kickoff);
   160→        return rank(a) === 2 ? kb - ka : ka - kb; // finished: newest first, others: earliest first
   161→      });
   162→    }
   163→    groups.sort((a, b) => {
   164→      const aLive = a.list.some((m) => m.status === "LIVE") ? 0 : 1;
   165→      const bLive = b.list.some((m) => m.status === "LIVE") ? 0 : 1;
   166→      if (aLive !== bLive) return aLive - bLive;
   167→      return +new Date(a.list[0].kickoff) - +new Date(b.list[0].kickoff);
   168→    });
   169→    return groups;
   170→  }, [pool]);
   171→
   172→  /** Right-rail quick list, resolved from competitions present in the data. */
   173→  const topCompetitions = useMemo(() => {
   174→    const byShort = new Map<string, Match["competition"]>();
   175→    for (const m of matches) byShort.set(m.competition.shortName, m.competition);
   176→    return TOP_COMPETITIONS.map((s) => byShort.get(s)).filter((c): c is Match["competition"] => Boolean(c));
   177→  }, [matches]);
   178→
   179→  const counts: Record<Tab, number> = {
   180→    all: dayMatches.length,
   181→    live: liveCount,
   182→    upcoming: dayMatches.filter((m) => m.status === "SCHEDULED").length,
   183→    finished: dayMatches.filter((m) => m.status === "FINISHED").length,
   184→  };
   185→
   186→  const todayLine = formatKickoff(new Date().toISOString()).date;
   187→
   188→  const emptyCopy =
   189→    tab === "live"
   190→      ? { title: "No live matches right now", hint: "Scores move to Live the moment a match kicks off." }
   191→      : tab === "upcoming"
   192→        ? { title: "No upcoming fixtures", hint: `Nothing is scheduled on ${selectedDay.label}. Try another date.` }
   193→        : tab === "finished"
   194→          ? { title: "No results yet", hint: `No matches have finished on ${selectedDay.label}.` }
   195→          : { title: `No matches on ${selectedDay.label}`, hint: "Pick another date from the rail above." };
   196→
   197→  return (
   198→    <div className="mx-auto w-full max-w-[1320px] px-4 pb-14 pt-5 sm:px-6 sm:pt-6">
   199→      <PageHeader
   200→        title="Live Scores"
   201→        kicker="match centre"
   202→        subtitle={`${todayLine} ${new Date().getFullYear()} — every fixture, live minute and final score`}
   203→        icon={<Zap className="h-5 w-5" aria-hidden />}
   204→      >
   205→        <span
   206→          className={cn(
   207→            "flex shrink-0 items-center gap-1.5 rounded-md border px-2.5 py-1.5 text-[10.5px] font-bold",
   208→            liveConnected
   209→              ? "border-primary/35 bg-primary/10 text-primary"
   210→              : "border-border bg-muted text-muted-foreground"
   211→          )}
   212→        >
   213→          <span
   214→            className={cn("h-1.5 w-1.5 rounded-[2px]", liveConnected ? "live-dot bg-primary" : "bg-muted-foreground")}
   215→            aria-hidden
   216→          />
   217→          {liveConnected ? "Real-time connected" : "Offline mode"}
   218→        </span>
   219→      </PageHeader>
   220→
   221→      {/* ── Sticky sub-nav: date rail + status tabs ─────────────── */}
   222→      <div
   223→        className="glass sticky top-14 z-30 -mx-4 border-b border-border sm:-mx-6 sm:top-[60px] lg:top-[99px]"
   224→        style={headerH != null ? { top: `${headerH}px` } : undefined}
   225→      >
   226→        <div className="flex flex-col gap-2 px-4 py-2.5 sm:px-6 lg:flex-row lg:items-center lg:justify-between lg:gap-4">
   227→          {/* Date rail */}
   228→          <div
   229→            className="no-scrollbar -mx-1 flex items-center gap-1 overflow-x-auto px-1"
   230→            role="group"
   231→            aria-label="Match date"
   232→          >
   233→            {dayChips.map((chip) => (
   234→              <button
   235→                key={chip.offset}
   236→                onClick={() => setDayOffset(chip.offset)}
   237→                aria-pressed={dayOffset === chip.offset}
   238→                title={chip.fullLabel}
   239→                className={cn(
   240→                  "flex shrink-0 items-center gap-1.5 rounded-lg px-2.5 py-1.5 text-[11.5px] font-semibold transition-colors",
   241→                  dayOffset === chip.offset
   242→                    ? "bg-elevated text-foreground"
   243→                    : "text-muted-foreground hover:bg-white/[0.04] hover:text-foreground"
   244→                )}
   245→              >
   246→                {chip.label}
   247→                {chip.count > 0 && (
   248→                  <span
   249→                    className={cn(
   250→                      "tabular rounded px-1 text-[9.5px] font-bold leading-[1.5]",
   251→                      dayOffset === chip.offset ? "bg-white/[0.09]" : "bg-white/[0.05] text-muted-foreground"
   252→                    )}
   253→                  >
   254→                    {fmtNum(chip.count)}
   255→                  </span>
   256→                )}
   257→                {chip.offset === 0 && liveCount > 0 && (
   258→                  <span className="live-dot h-1.5 w-1.5 rounded-full bg-fhb-red" aria-hidden />
   259→                )}
   260→              </button>
   261→            ))}
   262→          </div>
   263→
   264→          {/* Status filter tabs */}
   265→          <div
   266→            className="no-scrollbar flex shrink-0 items-center gap-0.5 self-start overflow-x-auto rounded-lg border border-border bg-muted/40 p-1 lg:self-auto"
   267→            role="tablist"
   268→            aria-label="Match status filter"
   269→          >
   270→            {STATUS_TABS.map((key) => (
   271→              <button
   272→                key={key}
   273→                role="tab"
   274→                aria-selected={tab === key}
   275→                onClick={() => setUserTab(key)}
   276→                className={cn(
   277→                  "flex shrink-0 items-center justify-center gap-1.5 rounded-md px-2.5 py-1.5 text-[11.5px] font-bold transition-colors sm:px-3",
   278→                  tab === key ? "bg-elevated text-foreground" : "text-muted-foreground hover:text-foreground"
   279→                )}
   280→              >
   281→                {key === "live" && liveCount > 0 && (
   282→                  <span
   283→                    className={cn("h-1.5 w-1.5 shrink-0 rounded-[2px] bg-fhb-red", tab !== "live" && "live-dot")}
   284→                    aria-hidden
   285→                  />
   286→                )}
   287→                {TAB_LABELS[key]}
   288→                <span
   289→                  className={cn(
   290→                    "tabular rounded px-1 text-[9.5px] font-bold leading-[1.5]",
   291→                    tab === key ? "bg-white/[0.09]" : "bg-white/[0.05] text-muted-foreground"
   292→                  )}
   293→                >
   294→                  {fmtNum(counts[key])}
   295→                </span>
   296→              </button>
   297→            ))}
   298→          </div>
   299→        </div>
   300→      </div>
   301→
   302→      {/* ── Match list + side rail ──────────────────────────────── */}
   303→      <div className="mt-6 grid gap-6 lg:grid-cols-12">
   304→        <div className="min-w-0 lg:col-span-8">
   305→          {isLoading ? (
   306→            <div className="space-y-4" aria-hidden>
   307→              {[0, 1].map((i) => (
   308→                <div key={i} className="overflow-hidden rounded-xl border border-border bg-card">
   309→                  <div className="border-b border-border bg-muted/40 px-3.5 py-3">
   310→                    <Skeleton className="h-4 w-44 rounded-md" />
   311→                  </div>
   312→                  <div className="divide-y divide-border/60">
   313→                    {[0, 1, 2].map((j) => (
   314→                      <div key={j} className="flex items-center gap-3 px-4 py-3">
   315→                        <Skeleton className="h-9 w-10 rounded-md" />
   316→                        <div className="flex-1 space-y-2">
   317→                          <Skeleton className="h-3.5 w-2/3 rounded-md" />
   318→                          <Skeleton className="h-3.5 w-1/2 rounded-md" />
   319→                        </div>
   320→                      </div>
   321→                    ))}
   322→                  </div>
   323→                </div>
   324→              ))}
   325→            </div>
   326→          ) : grouped.length === 0 ? (
   327→            <div className="reveal flex flex-col items-center justify-center gap-2.5 rounded-xl border border-border bg-card px-6 py-14 text-center">
   328→              <span className="flex h-12 w-12 items-center justify-center rounded-xl bg-muted text-muted-foreground">
   329→                {tab === "live" ? (
   330→                  <Activity className="h-5 w-5" aria-hidden />
   331→                ) : (
   332→                  <CalendarX className="h-5 w-5" aria-hidden />
   333→                )}
   334→              </span>
   335→              <p className="text-sm font-semibold text-foreground">{emptyCopy.title}</p>
   336→              <p className="max-w-[36ch] text-xs leading-relaxed text-muted-foreground">{emptyCopy.hint}</p>
   337→            </div>
   338→          ) : (
   339→            <div className="space-y-4">
   340→              {grouped.map((g) => {
   341→                const first = g.list[0];
   342→                const round = first.round ?? (first.matchday ? `Matchday ${fmtNum(first.matchday)}` : null);
   343→                const hasLive = g.list.some((m) => m.status === "LIVE");
   344→                return (
   345→                  <section
   346→                    key={g.comp.id}
   347→                    className="reveal overflow-hidden rounded-xl border border-border bg-card"
   348→                    aria-label={g.comp.name}
   349→                  >
   350→                    {/* Competition group header */}
   351→                    <button
   352→                      onClick={() => navigate({ name: "competition-detail", id: g.comp.id })}
   353→                      className="group flex w-full items-center gap-2.5 border-b border-border bg-muted/40 px-3.5 py-2.5 text-left transition-colors hover:bg-white/[0.03]"
   354→                      aria-label={`View competition: ${g.comp.name}`}
   355→                    >
   356→                      <span
   357→                        className="h-2 w-2 shrink-0 rounded-[3px]"
   358→                        style={{ backgroundColor: g.comp.accentColor ?? "var(--primary)" }}
   359→                        aria-hidden
   360→                      />
   361→                      <span className="min-w-0 truncate text-[12.5px] font-bold text-foreground" title={g.comp.name}>
   362→                        {g.comp.name}
   363→                      </span>
   364→                      {round && (
   365→                        <span className="hidden min-w-0 truncate text-[11px] font-medium text-muted-foreground md:inline">
   366→                          · {round}
   367→                        </span>
   368→                      )}
   369→                      {hasLive && (
   370→                        <span className="flex shrink-0 items-center gap-1 rounded bg-fhb-red/12 px-1.5 py-0.5 text-[9.5px] font-bold uppercase tracking-[0.08em] text-fhb-red">
   371→                          <span className="live-dot h-1 w-1 rounded-full bg-fhb-red" aria-hidden />
   372→                          Live
   373→                        </span>
   374→                      )}
   375→                      <span className="tabular ml-auto shrink-0 text-[10.5px] font-semibold text-muted-foreground">
   376→                        {fmtNum(g.list.length)} {g.list.length === 1 ? "match" : "matches"}
   377→                      </span>
   378→                      <ChevronRight
   379→                        className="h-3.5 w-3.5 shrink-0 text-muted-foreground transition-transform group-hover:translate-x-0.5"
   380→                        aria-hidden
   381→                      />
   382→                    </button>
   383→                    {/* Dense match rows */}
   384→                    <div className="divide-y divide-border/60">
   385→                      {g.list.map((m) => (
   386→                        <MatchRow key={m.id} match={m} showCompetition={false} />
   387→                      ))}
   388→                    </div>
   389→                  </section>
   390→                );
   391→              })}
   392→            </div>
   393→          )}
   394→
   395→          {/* In-feed ad after the match list */}
   396→          <div className="reveal mt-6">
   397→            <AdSlot placement="IN_FEED" />
   398→          </div>
   399→        </div>
   400→
   401→        {/* ── Side rail ─────────────────────────────────────────── */}
   402→        <aside className="min-w-0 space-y-4 lg:col-span-4" aria-label="Scores side rail">
   403→          {liveCount > 0 && (
   404→            <div className="reveal overflow-hidden rounded-xl border border-fhb-red/30 bg-card">
   405→              <p className="flex items-center gap-2 border-b border-border bg-fhb-red/[0.07] px-4 py-3 text-[11px] font-bold uppercase tracking-[0.1em] text-fhb-red">
   406→                <span className="live-dot h-1.5 w-1.5 rounded-full bg-fhb-red" aria-hidden />
   407→                Live Now — {fmtNum(liveCount)}
   408→              </p>
   409→              <div className="divide-y divide-border/60">
   410→                {liveMatches.slice(0, 4).map((m) => (
   411→                  <MatchRow key={m.id} match={m} showCompetition />
   412→                ))}
   413→              </div>
   414→              <button
   415→                onClick={() => setUserTab("live")}
   416→                className="flex w-full items-center justify-center gap-1 border-t border-border py-2.5 text-[11px] font-bold text-primary transition-colors hover:bg-primary/5"
   417→              >
   418→                All live matches <ChevronRight className="h-3 w-3" aria-hidden />
   419→              </button>
   420→            </div>
   421→          )}
   422→
   423→          <div className="reveal">
   424→            <AdSlot placement="SIDEBAR" />
   425→          </div>
   426→
   427→          {isLoading ? (
   428→            <Skeleton className="h-[236px] rounded-xl" />
   429→          ) : (
   430→            topCompetitions.length > 0 && (
   431→              <div className="reveal overflow-hidden rounded-xl border border-border bg-card">
   432→                <p className="flex items-center gap-2 border-b border-border bg-muted/40 px-4 py-3 text-[11px] font-bold uppercase tracking-[0.1em] text-foreground">
   433→                  <Trophy className="h-3.5 w-3.5 text-primary" aria-hidden />
   434→                  Top Competitions
   435→                </p>
   436→                <div className="grid grid-cols-2 gap-2 p-3">
   437→                  {topCompetitions.map((c) => (
   438→                    <button
   439→                      key={c.id}
   440→                      onClick={() => navigate({ name: "competition-detail", id: c.id })}
   441→                      title={c.name}
   442→                      className="group flex items-center gap-2 rounded-lg border border-border bg-elevated/50 px-2.5 py-2 text-left transition-colors hover:border-primary/40"
   443→                    >
   444→                      <span
   445→                        className="h-2 w-2 shrink-0 rounded-[3px]"
   446→                        style={{ backgroundColor: c.accentColor ?? "var(--primary)" }}
   447→                        aria-hidden
   448→                      />
   449→                      <span className="min-w-0 truncate text-[11.5px] font-bold text-foreground/90 transition-colors group-hover:text-primary">
   450→                        {c.shortName}
   451→                      </span>
   452→                    </button>
   453→                  ))}
   454→                </div>
   455→              </div>
   456→            )
   457→          )}
   458→        </aside>
   459→      </div>
   460→    </div>
   461→  );
   462→}

   464→/* ════════════════════════════════════════════════════════════════
   465→   MatchDetailView — broadcast match centre
   466→   Scoreboard hero, event timeline, match info, countdown for fixtures.
   467→   ════════════════════════════════════════════════════════════════ */
   468→
   469→export function MatchDetailView({ matchId }: { matchId: string }) {

   927→function eventIcon(type: string, className?: string) {
   928→  switch (type) {
   929→    case "GOAL":

   931→    case "YELLOW":
   932→      return <span className="h-3.5 w-2.5 rounded-[2px] bg-yellow-400/90" aria-label="Yellow card" />;
   933→    case "RED":
