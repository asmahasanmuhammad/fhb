   173→   173→

   175→   175→    queryKey: ["admin", "news"],
   176→   176→    queryFn: async () => {

   179→   179→    },
   180→   180→  });
   181→   181→
   182→   182→  const save = useMutation({
   183→   183→    mutationFn: async () => {
   184→   184→      const isEdit = !!editing;
   185→   185→      const res = await fetch("/api/admin/news", {
   186→   186→        method: isEdit ? "PUT" : "POST",
   187→   187→        headers: { "Content-Type": "application/json" },
   188→   188→        body: JSON.stringify(isEdit ? { id: editing!.id, ...form } : form),
   189→   189→      });
   190→   190→      const json = await res.json();
   191→   191→      if (!json.success) throw new Error(json.error);
   192→   192→      return json;
   193→   193→    },
   194→   194→    onSuccess: () => {

   196→   196→      setDialogOpen(false);
   197→   197→      setEditing(null);
   198→   198→      queryClient.invalidateQueries({ queryKey: ["admin", "news"] });
   199→   199→      queryClient.invalidateQueries({ queryKey: ["news"] });
   200→   200→    },
   201→   201→    onError: (e: Error) => toast({ title: "Save failed", description: e.message, variant: "destructive" }),

   230→   230→
   231→   231→  const articles = data?.articles ?? [];
   232→   232→  const inputCls = "w-full rounded-xl border border-border bg-background px-3.5 py-2.5 text-sm outline-none transition-all focus:border-primary/60 focus:ring-2 focus:ring-primary/20";
   233→   233→
