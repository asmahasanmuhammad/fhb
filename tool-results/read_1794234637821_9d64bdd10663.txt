   167→function NewsManager() {
   168→  const { toast } = useToast();
   169→  const queryClient = useQueryClient();
   170→  const [dialogOpen, setDialogOpen] = useState(false);
   171→  const [editing, setEditing] = useState<Article | null>(null);
   172→  const [form, setForm] = useState({ title: "", excerpt: "", content: "", category: "BFL", coverUrl: "", featured: false, status: "PUBLISHED" });
   173→

   175→    queryKey: ["admin", "news"],
   176→    queryFn: async () => {
   177→      const res = await fetch("/api/admin/news");
   178→      return res.json().then((j) => j.data);
   179→    },
   180→  });
   181→
   182→  const save = useMutation({
   183→    mutationFn: async () => {
   184→      const isEdit = !!editing;
   185→      const res = await fetch("/api/admin/news", {
   186→        method: isEdit ? "PUT" : "POST",
   187→        headers: { "Content-Type": "application/json" },
   188→        body: JSON.stringify(isEdit ? { id: editing!.id, ...form } : form),
   189→      });
   190→      const json = await res.json();
   191→      if (!json.success) throw new Error(json.error);
   192→      return json;
   193→    },

   195→      toast({ title: editing ? "Article updated" : "Article published" });

   197→      setEditing(null);

   205→    mutationFn: async (id: string) => {

   245→
   246→      {isLoading ? (
   247→        [1, 2, 3].map((i) => <Skeleton key={i} className="mb-2 h-16 rounded-lg" />)
   248→      ) : (
   249→        <div className="edge-lit overflow-hidden rounded-2xl border border-border bg-card">
   250→          <div className="divide-y divide-border/50">
   251→            {articles.map((a) => (
   252→              <div key={a.id} className="flex flex-wrap items-center gap-3 p-3 transition-colors hover:bg-accent/30">

   294→          <div className="space-y-3.5">
   295→            <div>

   391→type AdminMatch = {
   392→  id: string; status: string; homeScore: number | null; awayScore: number | null; kickoff: string; round: string | null; venue: string | null;
   393→  homeTeam: { id: string; name: string; initials: string; color: string };
   394→  awayTeam: { id: string; name: string; initials: string; color: string };
   395→  competition: { id: string; name: string; shortName: string };
   396→};
   397→
   398→function MatchManager() {
   399→  const { toast } = useToast();
   400→  const queryClient = useQueryClient();
   401→
   402→  const { data, isLoading } = useQuery<{ matches: AdminMatch[] }>({
   403→    queryKey: ["admin", "matches"],
   404→    queryFn: async () => {
   405→      const res = await fetch("/api/admin/matches");
   406→      return res.json().then((j) => j.data);
   407→    },
   408→  });
   409→
   410→  const update = useMutation({
   411→    mutationFn: async (payload: Record<string, unknown>) => {
   412→      const res = await fetch("/api/admin/matches", {
   413→        method: "PUT",
   414→        headers: { "Content-Type": "application/json" },
   415→        body: JSON.stringify(payload),
   416→      });
   417→      const json = await res.json();
   418→      if (!json.success) throw new Error(json.error);
   419→      return json;
   420→    },
   421→    onSuccess: (_d, payload) => {
   422→      const msg =
   423→        payload.status === "LIVE"
   424→          ? "Match is now live! The live simulation service will pick it up within 15 seconds."
   425→          : "Match updated";
   426→      toast({ title: msg });
   427→      queryClient.invalidateQueries({ queryKey: ["admin", "matches"] });
   428→      queryClient.invalidateQueries({ queryKey: ["matches"] });
   429→    },
   430→    onError: (e: Error) => toast({ title: "Update failed", description: e.message, variant: "destructive" }),
   431→  });
   432→
   433→  const matches = data?.matches ?? [];
   434→  const sorted = [...matches].sort((a, b) => {
   435→    const rank = (s: string) => (s === "LIVE" ? 0 : s === "SCHEDULED" ? 1 : 2);
   436→    return rank(a.status) - rank(b.status) || +new Date(b.kickoff) - +new Date(a.kickoff);
   437→  });
   438→
   439→  return (
   440→    <div>
   441→      <p className="mb-3 flex items-start gap-2.5 rounded-xl border border-fhb-red/25 bg-fhb-red/5 p-3.5 text-[11px] leading-relaxed text-muted-foreground">
   442→        <Info className="mt-0.5 h-4 w-4 shrink-0 text-fhb-red" aria-hidden />
